cmake_minimum_required(VERSION 3.22)
project(naturegame CXX C)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -O3 -ffast-math -fvisibility=hidden -Wno-unused-parameter")
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} -O3")
set(GLUE_DIR ${ANDROID_NDK}/sources/android/native_app_glue)
add_library(app_glue STATIC ${GLUE_DIR}/android_native_app_glue.c)
target_include_directories(app_glue PUBLIC ${GLUE_DIR})
add_library(game SHARED main.cpp game.cpp game_ui.cpp game_survival.cpp game_animals.cpp renderer.cpp chunks.cpp world.cpp pak.cpp)
target_include_directories(game PRIVATE ${GLUE_DIR})
set_target_properties(game PROPERTIES LINK_FLAGS "-u ANativeActivity_onCreate -Wl,--gc-sections")
target_link_libraries(game app_glue android EGL GLESv3 log)
